# ─────────────────────────────────────────────────────────────────────────────
# Reglas para AGREGAR al .htaccess existente de redone.agency (Apache).
# No reemplaces tu .htaccess: pega este bloque ARRIBA de las reglas que ya tengas,
# porque las redirecciones deben ejecutarse antes que cualquier reescritura del sitio.
# Después de subirlo, prueba con:
#   curl -I https://www.redone.agency/      → 301 a https://redone.agency/
#   curl -I https://redone.agency/index.php → 301 a https://redone.agency/
#   curl -I https://redone.agency/blog/     → 301 a https://redone.agency/blog
# ─────────────────────────────────────────────────────────────────────────────

<IfModule mod_rewrite.c>
  RewriteEngine On

  # 1) www → sin www (hoy www.redone.agency responde 200 y duplica todo el sitio)
  RewriteCond %{HTTP_HOST} ^www\.redone\.agency$ [NC]
  RewriteRule ^ https://redone.agency%{REQUEST_URI} [L,R=301,NE]

  # 2) /index.php → /   (THE_REQUEST evita bucles con DirectoryIndex)
  RewriteCond %{THE_REQUEST} \s/+index\.php[\s?] [NC]
  RewriteRule ^index\.php$ / [L,R=301]

  # 3) /blog/ → /blog   (hoy /blog/ da 404)
  RewriteRule ^blog/$ /blog [L,R=301]
</IfModule>

# Encabezados de seguridad (no afectan el diseño)
<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set X-Frame-Options "SAMEORIGIN"
</IfModule>

# Caché larga para imágenes, fuentes, CSS y JS
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/webp "access plus 1 year"
  ExpiresByType image/jpeg "access plus 1 year"
  ExpiresByType image/png "access plus 1 year"
  ExpiresByType image/svg+xml "access plus 1 year"
  ExpiresByType font/woff2 "access plus 1 year"
  ExpiresByType text/css "access plus 1 month"
  ExpiresByType application/javascript "access plus 1 month"
  ExpiresByType video/mp4 "access plus 1 year"
</IfModule>

# Compresión
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml text/plain application/xml
</IfModule>
