# ─────────────────────────────────────────────────────────────────────────────
# .htaccess de redone.agency. Este archivo REEMPLAZA completo al actual.
# El orden importa: primero las redirecciones 301 y después las reescrituras
# internas (blog, /en, términos). En el archivo anterior las redirecciones
# estaban al final y no se ejecutaban (www.redone.agency seguía dando 200).
# ─────────────────────────────────────────────────────────────────────────────

Options -Indexes
DirectoryIndex index.php index.html

<IfModule mod_rewrite.c>
RewriteEngine On

# ── 1. Redirecciones 301 (una URL por página) ───────────────────────────────

# www → sin www
RewriteCond %{HTTP_HOST} ^www\.redone\.agency$ [NC]
RewriteRule ^ https://redone.agency%{REQUEST_URI} [L,R=301,NE]

# (http → https ya lo hace el servidor; no se repite aquí para no arriesgar un bucle)

# Archivos internos pedidos directamente → URL limpia.
# THE_REQUEST solo mira lo que pidió el navegador, así no choca con las reescrituras de abajo.
RewriteCond %{THE_REQUEST} \s/+index\.php[\s?] [NC]
RewriteRule ^index\.php$ https://redone.agency/ [L,R=301]
RewriteCond %{THE_REQUEST} \s/+index_en\.php[\s?] [NC]
RewriteRule ^index_en\.php$ https://redone.agency/en [L,R=301]
RewriteCond %{THE_REQUEST} \s/+templates/terminos\.php[\s?] [NC]
RewriteRule ^templates/terminos\.php$ https://redone.agency/terminos [L,R=301]
RewriteCond %{THE_REQUEST} \s/+templates/descarga\.php[\s?] [NC]
RewriteRule ^templates/descarga\.php$ https://redone.agency/descarga [L,R=301]
RewriteCond %{THE_REQUEST} \s/+templates/noticiassec/index\.php[\s?] [NC]
RewriteRule ^templates/noticiassec/index\.php$ https://redone.agency/blog [L,R=301]
RewriteCond %{THE_REQUEST} \s/+templates/noticiassec/index_en\.php[\s?] [NC]
RewriteRule ^templates/noticiassec/index_en\.php$ https://redone.agency/en/blog [L,R=301]

# Sin diagonal final (hoy /blog/ y /blog/articulo/ dan 404, y /en/ duplica /en)
RewriteRule ^(blog|en|en/blog|terminos|descarga)/$ https://redone.agency/$1 [L,R=301]
RewriteRule ^(blog|en/blog)/([a-z0-9-]+)/$ https://redone.agency/$1/$2 [L,R=301]

# Atajo para agendar
RewriteRule ^book-a-call/?$ https://calendar.google.com/appointments/schedules/AcZssZ1ZDJV0jsraNBK5FDkcJqTJto8cr4IQEfbv9_0QbTwlFFauDt-fm47cANCUR0i68WkxTHBwiIMD [L,R=301]

# ── 2. Reescrituras internas (no cambian la URL que ve el usuario) ──────────

# Secciones
RewriteRule ^terminos$ /templates/terminos.php [B,L]
RewriteRule ^descarga$ /templates/descarga.php [B,L]

# Blog
RewriteRule ^blog$ /templates/noticiassec/index.php [B,L]
RewriteRule ^blog/([a-z0-9-]+)$ /templates/noticiassec/individual.php?slug=$1 [B,L]

# Inglés
RewriteRule ^en$ /index_en.php [B,L]
RewriteRule ^en/blog$ /templates/noticiassec/index_en.php [B,L]
RewriteRule ^en/blog/([a-z0-9-]+)$ /templates/noticiassec/individual_en.php?slug=$1 [B,L]
</IfModule>

# ── 3. Seguridad ────────────────────────────────────────────────────────────
<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=()"
</IfModule>

# Nadie debe poder descargar estos archivos
<FilesMatch "(^\.|\.(env|ini|log|sql|bak|old|sh)$|composer\.(json|lock))">
  Require all denied
</FilesMatch>

# ── 4. Caché ────────────────────────────────────────────────────────────────
# Imágenes, video y fuentes: 1 año (si cambian una imagen, súbanla con otro nombre).
# CSS y JS: 1 semana. Antes tenían 1 año con "immutable": quien ya visitó el sitio
# no vería los cambios de style.css hasta dentro de un año.
# Si algún día versionan los archivos (style.css?v=2), pueden subirlo a 1 año.
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/webp "access plus 1 year"
  ExpiresByType image/avif "access plus 1 year"
  ExpiresByType image/jpeg "access plus 1 year"
  ExpiresByType image/png "access plus 1 year"
  ExpiresByType image/gif "access plus 1 year"
  ExpiresByType image/svg+xml "access plus 1 year"
  ExpiresByType image/x-icon "access plus 1 year"
  ExpiresByType video/mp4 "access plus 1 year"
  ExpiresByType video/MP2T "access plus 1 year"
  ExpiresByType font/woff2 "access plus 1 year"
  ExpiresByType font/woff "access plus 1 year"
  ExpiresByType text/css "access plus 1 week"
  ExpiresByType application/javascript "access plus 1 week"
  ExpiresByType text/javascript "access plus 1 week"
  ExpiresByType application/x-mpegURL "access plus 1 day"
  ExpiresByType text/html "access plus 0 seconds"
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.(webp|avif|jpe?g|png|gif|svg|ico|mp4|ts|woff2?)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "\.(css|js)$">
    Header set Cache-Control "public, max-age=604800"
  </FilesMatch>
  <FilesMatch "\.m3u8$">
    Header set Cache-Control "public, max-age=86400"
  </FilesMatch>
</IfModule>

# ── 5. Compresión ───────────────────────────────────────────────────────────
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css text/xml application/xml application/xhtml+xml
  AddOutputFilterByType DEFLATE application/javascript text/javascript application/json image/svg+xml application/x-mpegURL
</IfModule>
